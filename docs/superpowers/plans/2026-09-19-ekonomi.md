```ts
/**
 * Transfer bedeli. Üsteldir, çünkü bir sürücünün değeri yeteneğiyle doğrusal
 * artmaz: ort 95 bir yıldız, ort 76 bir sürücünün dört katı eder. Bu eğri iki
 * işi birden yapar — yıldızı yarış parası biriktirerek alınamaz kılar (13
 * yarışlık gelir) ve ticareti kârlı yapar, çünkü 62'den 76'ya çıkarmak değeri
 * ikiye katlar. Potansiyel primi bilerek küçüktür (25/puan): pazarın
 * potansiyeli ucuza satması, "ucuza al, geliştir, sat" döngüsünün kaynağıdır.
 */
export function driverFee(driver: Driver): number {
  const overall = overallOf(driver.stats);
  const upside = Math.max(0, driver.potential - overall);
  return Math.round(800 * Math.pow(1.075, overall - 55) + upside * 25);
}

/** Menajer komisyonu: satışta değerin %20'si kesilir. */
export const SALE_COMMISSION = 0.2;

/** Bir sürücüyü sattığında eline geçen. */
export const saleValue = (driver: Driver): number =>
  Math.round(driverFee(driver) * (1 - SALE_COMMISSION));
```

export function crippleSetup(setup: CarSetup, buildingLabel: string | undefined): CarSetup {

import { ECONOMY_SCALE } from './economy';

/** `level` = MEVCUT seviye; dönen değer bir üst seviyenin fiyatı. */

export interface FactoryEffects {

export const CAR_BASELINE = 55;

/**

export function regressCar(stat: number, factoryFloorBonus: number): number {

// ── Sezon simülasyonu ──────────────────────────────────────────────────────
// Gerçek formülleri kullanır; yarış motorunu değil (o ayrı bir iş). Amaç
// ekonominin temposunu ölçmek: para ve takvim bir sezonda nereye götürüyor.
interface SimOut { stats: number[]; bought: number[]; builds: number; pos: number; rp: number }

function simulateSeason(wagePerRace: number, reserveRp: number): SimOut {
  const ROUNDS = 23, RACE_MS = 60 * 3600_000;
  let rp = 3000, pos = 6, builds = 0;
  const stats = [67, 58, 72], bought = [0, 0, 0];
  for (let r = 1; r <= ROUNDS; r++) {
    rp += Math.round(1450 - 70 * (pos - 1)) - wagePerRace;
    let ms = RACE_MS;
    // Tek tezgah: yarış arasına sığdığı ve padok rezervi bozulmadığı sürece
    // en düşük stat'a yatır.
    for (;;) {
      const i = stats.indexOf(Math.min(...stats));
      const cost = upgradeCostFor(bought[i]), dur = upgradeDurationMs(bought[i]);
      if (rp - cost < reserveRp || dur > ms) break;
      rp -= cost; ms -= dur; builds++;
      stats[i] = Math.min(100, stats[i] + 6); bought[i]++;
    }
    const avg = stats.reduce((a, b) => a + b, 0) / 3;
    pos = Math.max(1, Math.min(11, Math.round(11 - (avg - 55) / 4.2)));
  }
  return { stats, bought, builds, pos, rp };
}
